function noLinks(p){
  for(let q=path.resolve(p);;q=path.dirname(q)){
    const s=fs.lstatSync(q); ensure(s.isDirectory()&&!s.isSymbolicLink(),'nonregular_parent');
    ensure(fs.realpathSync.native(q).replace(/^\\\\\?\\/,'').toLowerCase()===q.toLowerCase(),'aliased_parent');
    if(path.dirname(q)===q)break;
  }
}

function observe(env,root){

async function reserve(){
  const server=net.createServer(); await new Promise((resolve,reject)=>{
    server.once('error',reject);server.listen({host:'127.0.0.1',port:0,exclusive:true},resolve);
  });
  const port=server.address().port;
  if([4317,5173,7437,7438].includes(port)||ports.some(p=>p.port===port)){
    await new Promise(r=>server.close(r));ensure(false,'port_selection_refused');
  }
  return {server,port};
}
function health(port){
  // No redirects; no headers, tokens, raw response bodies or arbitrary fields retained.
  return new Promise(resolve=>{
    let done=false,timer;const finish=value=>{if(!done){done=true;clearTimeout(timer);resolve(value);}};
    const request=http.get({host:'127.0.0.1',port,path:'/health',agent:false,timeout:1500},response=>{
      let count=0;const parts=[];
      response.on('data',part=>{count+=part.length;if(count>4096){response.destroy();finish(null);}else parts.push(part);});
      response.on('error',()=>finish(null));
      response.on('end',()=>{
        try{const body=JSON.parse(Buffer.concat(parts).toString('utf8'));
          const allowed=['starting','restarting','error','needs_you','idle','working','thinking'];
          finish(response.statusCode===200&&body.ok===true&&allowed.includes(body.agentState)
            ?{ok:true,agentState:body.agentState}:null);
        }catch{finish(null);}
      });
    });
    timer=setTimeout(()=>{request.destroy();finish(null);},1500);
    request.once('timeout',()=>{request.destroy();finish(null);});request.once('error',()=>finish(null));
  });
}
